\fbox{\large\textbf{1. Mathematical Tools}}
\vspace{0.5mm}

\textbf{Differentiation Rules:}
\begin{itemize}
\item Product: $(uv)' = u'v + uv'$
\item Quotient: $(\frac{u}{v})' = \frac{u'v - uv'}{v^2}$
\item Chain: $\frac{d}{dx}f(g(x)) = f'(g(x))g'(x)$
\item Logarithm: $\frac{d}{dx}\ln(f(x)) = \frac{f'(x)}{f(x)}$
\item Exponential: $\frac{d}{dx}e^{f(x)} = f'(x)e^{f(x)}$
\end{itemize}

\textbf{Integration by Parts:}
$$\int u \, dv = uv - \int v \, du$$
\begin{itemize}
\item Choice order (L.I.A.T.E): Log, Inverse trig, Algebraic, Trig, Exponential
\end{itemize}

\textbf{Gamma Function:}
\begin{itemize}
\item $\Gamma(\alpha) = \int_0^\infty x^{\alpha-1}e^{-x} dx$
\item Property: $\Gamma(\alpha) = (\alpha-1)\Gamma(\alpha-1)$
\item Integer: $\Gamma(n) = (n-1)!$
\end{itemize}

\vspace{2mm}
\fbox{\large\textbf{1.5 Key Transformations}}
\vspace{0.5mm}

\textbf{Transformation Formula (Univariate)}

If $Y = g(X)$ is monotonic:
$$f_Y(y) = f_X(g^{-1}(y))\left|\frac{dx}{dy}\right|$$

For many-to-one (e.g., $Y = X^2$), sum over roots

\textbf{Jacobian (Bivariate)}

For $(U,V) = g(X,Y)$:
$$f_{U,V}(u,v) = f_{X,Y}(x,y) |J|$$

where $J = \det\left[\begin{matrix}\frac{\partial x}{\partial u} & \frac{\partial x}{\partial v} \\ \frac{\partial y}{\partial u} & \frac{\partial y}{\partial v}\end{matrix}\right]$

\textbf{Order Statistics}

For i.i.d. $X_1, \ldots, X_n$ with CDF $F$, PDF $f$:

\textit{Minimum:} $f_{X_{(1)}}(x) = n[1-F(x)]^{n-1} f(x)$

\textit{Maximum:} $f_{X_{(n)}}(x) = n[F(x)]^{n-1} f(x)$

\textit{k-th:} $f_{X_{(k)}}(x) = \frac{n!}{(k-1)!(n-k)!} [F(x)]^{k-1} [1-F(x)]^{n-k} f(x)$

\textbf{Sums of Independent RVs:}
\begin{itemize}
\item $X_i \sim \text{Exp}(\lambda)$ iid: $\sum X_i \sim \text{Gamma}(n, \lambda)$
\item $X_i \sim \text{Poisson}(\lambda_i)$: $\sum X_i \sim \text{Poisson}(\sum \lambda_i)$
\item $X_i \sim N(\mu_i, \sigma_i^2)$: $\sum a_i X_i \sim N(\sum a_i\mu_i, \sum a_i^2\sigma_i^2)$
\item $X_i \sim \text{Binomial}(n_i, p)$: $\sum X_i \sim \text{Binomial}(\sum n_i, p)$
\end{itemize}

\vspace{2mm}
\fbox{\large\textbf{2. Sampling Distributions}}
\vspace{0.5mm}

\textbf{Law of Large Numbers (LLN)}

If $X_1, X_2, \ldots$ i.i.d. with $\E[X_i] = \mu$, $\Var(X_i) = \sigma^2 < \infty$:
$$\bar{X}_n = \frac{1}{n}\sum_{i=1}^n X_i \xrightarrow{P} \mu$$

Chebyshev proof: $P(|\bar{X}_n - \mu| > \varepsilon) \le \frac{\sigma^2/n}{\varepsilon^2} \to 0$

\textbf{Central Limit Theorem (CLT)}

If $X_i$ i.i.d. with mean $\mu$ and variance $\sigma^2 > 0$:
$$\frac{\bar{X}_n - \mu}{\sigma/\sqrt{n}} \xrightarrow{d} N(0,1)$$

Equivalently: $\bar{X}_n \approx N(\mu, \sigma^2/n)$ for large $n$

For sums: $S_n = \sum X_i \approx N(n\mu, n\sigma^2)$

\textbf{Continuity Correction (Discrete → Continuous)}

For $X \sim B(n,p)$: $P(X \le k) \approx P\left(Z \le \frac{k+0.5-np}{\sqrt{np(1-p)}}\right)$

Rule: Normal approx reliable if $np(1-p) \gtrsim 10$

\textbf{Chi-Square Distribution: $\chi^2(k)$}

If $Z_1, \ldots, Z_k \sim N(0,1)$ independent:
$$\sum_{i=1}^k Z_i^2 \sim \chi^2(k)$$

PDF: $f(x) = \frac{1}{2^{k/2}\Gamma(k/2)} x^{k/2-1} e^{-x/2}$, $x > 0$

$\E[\chi^2(k)] = k$, $\Var(\chi^2(k)) = 2k$

MGF: $M(t) = (1-2t)^{-k/2}$, $t < 1/2$

\textbf{Additivity:} If $X \sim \chi^2(k_1)$, $Y \sim \chi^2(k_2)$ independent:
$$X + Y \sim \chi^2(k_1 + k_2)$$

\textbf{t-Distribution: $t(n)$}

If $Z \sim N(0,1)$, $V \sim \chi^2(n)$ independent:
$$T = \frac{Z}{\sqrt{V/n}} \sim t(n)$$

$\E[T] = 0$ ($n > 1$), $\Var(T) = \frac{n}{n-2}$ ($n > 2$)

As $n \to \infty$: $t(n) \to N(0,1)$

\textbf{F-Distribution: $F(n_1, n_2)$}

If $V_1 \sim \chi^2(n_1)$, $V_2 \sim \chi^2(n_2)$ independent:
$$F = \frac{V_1/n_1}{V_2/n_2} \sim F(n_1, n_2)$$

$\E[F] = \frac{n_2}{n_2-2}$ ($n_2 > 2$)

If $F \sim F(n_1, n_2)$: $\frac{1}{F} \sim F(n_2, n_1)$

If $T \sim t(n)$: $T^2 \sim F(1, n)$

\textbf{Normal Sample: Key Results}

If $X_1, \ldots, X_n \sim N(\mu, \sigma^2)$ i.i.d.:
\begin{itemize}
\item $\bar{X} \sim N(\mu, \sigma^2/n)$
\item $\frac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)$ where $S^2 = \frac{1}{n-1}\sum(X_i-\bar{X})^2$
\item $\bar{X}$ and $S^2$ are independent
\item $\frac{\bar{X}-\mu}{\sigma/\sqrt{n}} \sim N(0,1)$ ($\sigma$ known)
\item $\frac{\bar{X}-\mu}{S/\sqrt{n}} \sim t(n-1)$ ($\sigma$ unknown)
\end{itemize}

\textbf{Delta Method}

If $\sqrt{n}(\hat{\theta}_n - \theta) \Rightarrow N(0, \tau^2)$ and $g'(\theta) \neq 0$:
$$\sqrt{n}(g(\hat{\theta}_n) - g(\theta)) \Rightarrow N(0, [g'(\theta)]^2 \tau^2)$$

Variance: $\Var(g(\hat{\theta}_n)) \approx [g'(\theta)]^2 \Var(\hat{\theta}_n)$

\textbf{Key Distribution Relationships:}
\begin{itemize}
\item $Z^2 \sim \chi^2(1)$ if $Z \sim N(0,1)$
\item $\chi^2(k_1) + \chi^2(k_2) = \chi^2(k_1+k_2)$ (indep)
\item $T^2 \sim F(1,n)$ if $T \sim t(n)$
\item $1/F \sim F(n_2, n_1)$ if $F \sim F(n_1, n_2)$
\item As $n \to \infty$: $t(n) \to N(0,1)$
\end{itemize}

\vspace{2mm}
\fbox{\large\textbf{3. Point Estimation}}
\vspace{0.5mm}

\textbf{Estimator Properties}

\textit{Bias:} $\text{Bias}(\hat{\theta}) = \E[\hat{\theta}] - \theta$

\textit{Variance:} $\Var(\hat{\theta}) = \E[(\hat{\theta} - \E[\hat{\theta}])^2]$

\textit{MSE:} $\text{MSE}(\hat{\theta}) = \Var(\hat{\theta}) + [\text{Bias}(\hat{\theta})]^2$

\textbf{Key Result:} Unbiased $\Rightarrow$ MSE $= \Var$

\textbf{Consistency}

$\hat{\theta}_n \xrightarrow{P} \theta$ as $n \to \infty$ (converges in probability)

\textit{Sufficient conditions:}
\begin{itemize}
\item Unbiased + $\Var(\hat{\theta}_n) \to 0$
\item $\text{MSE}(\hat{\theta}_n) \to 0$
\item $\text{Bias}(\hat{\theta}_n) \to 0$ and $\Var(\hat{\theta}_n) \to 0$
\end{itemize}

\textbf{Relative Efficiency}

For unbiased $\hat{\theta}_1$, $\hat{\theta}_2$:
$$\text{eff}(\hat{\theta}_1, \hat{\theta}_2) = \frac{\Var(\hat{\theta}_2)}{\Var(\hat{\theta}_1)}$$

eff $> 1$: $\hat{\theta}_1$ more efficient

\textbf{Common Estimator Biases}

\begin{itemize}
\item Sample variance (MLE): $\hat{\sigma}^2_{\text{MLE}} = \frac{1}{n}\sum(X_i-\bar{X})^2$ is \textbf{biased}
  \begin{itemize}
  \item $\E[\hat{\sigma}^2_{\text{MLE}}] = \frac{n-1}{n}\sigma^2$ (underestimates)
  \item Unbiased version: $S^2 = \frac{1}{n-1}\sum(X_i-\bar{X})^2$
  \end{itemize}
\item Uniform$(0, \theta)$: $\hat{\theta} = X_{(n)}$ is biased
  \begin{itemize}
  \item $\E[X_{(n)}] = \frac{n}{n+1}\theta$ (underestimates)
  \item $\E[X_{(1)}] = \frac{\theta}{n+1}$ (min)
  \item Unbiased: $\frac{n+1}{n}X_{(n)}$
  \end{itemize}
\end{itemize}

\textbf{Variance Formulas:}
\begin{itemize}
\item Linear combo: $\Var(aX + bY) = a^2\Var(X) + b^2\Var(Y) + 2ab\Cov(X,Y)$
\item Independent: $\Var(X + Y) = \Var(X) + \Var(Y)$
\item General: $\Var(X) = \E[X^2] - (\E[X])^2$
\end{itemize}

\textbf{Method of Moments (MoM)}

Match sample moments to population moments:
\begin{enumerate}
\item Set $\bar{X} = \E[X]$ (first moment)
\item Set $\frac{1}{n}\sum X_i^2 = \E[X^2]$ (second moment)
\item Solve for $\theta$
\end{enumerate}

\textit{Example:} Gamma$(\alpha, \beta)$:
\begin{itemize}
\item $\E[X] = \alpha\beta = \bar{X}$
\item $\Var(X) = \alpha\beta^2 = S^2$
\item Solve: $\hat{\beta} = S^2/\bar{X}$, $\hat{\alpha} = \bar{X}^2/S^2$
\end{itemize}

\textbf{Common Pitfalls:}
\begin{itemize}
\item $S^2$ (unbiased) has $n-1$; MLE $\hat{\sigma}^2$ has $n$
\item t-distribution df is $n-1$ (not $n$!)
\item For $\chi^2$ variance test: $(n-1)S^2/\sigma^2$
\item Type I error $= \alpha$, Type II error $= \beta$
\end{itemize}

\textbf{Maximum Likelihood (MLE)}

Likelihood: $L(\theta) = \prod_{i=1}^n f(x_i; \theta)$

Log-likelihood: $\ell(\theta) = \sum_{i=1}^n \log f(x_i; \theta)$

MLE: $\hat{\theta}_{\text{MLE}} = \arg\max_{\theta} \ell(\theta)$

Finding MLE: Solve $\frac{d\ell}{d\theta} = 0$ (score equation)

\textbf{Common MLEs:}
\begin{itemize}
\item Bernoulli$(p)$: $\hat{p} = \bar{X}$
\item Poisson$(\lambda)$: $\hat{\lambda} = \bar{X}$
\item Normal$(\mu, \sigma^2)$: $\hat{\mu} = \bar{X}$, $\hat{\sigma}^2 = \frac{1}{n}\sum(X_i-\bar{X})^2$
\item Exponential$(\lambda)$: $\hat{\lambda} = \frac{1}{\bar{X}}$
\item Uniform$(0, \theta)$: $\hat{\theta} = X_{(n)} = \max\{X_1, \ldots, X_n\}$
\end{itemize}

\textbf{MLE Finding Tips:}
\begin{itemize}
\item \textit{Support depends on $\theta$:} Uniform$(0,\theta)$ or $(a,b)$
  \begin{itemize}
  \item MLE is at boundary: $\max\{X_i\}$ or $\{\min\{X_i\}, \max\{X_i\}\}$
  \item \textbf{Don't differentiate!}
  \end{itemize}
\item \textit{Log tricks:} $\frac{d}{d\theta}\log f = \frac{f'}{f}$, $\log(a^b) = b\log a$
\item Check second derivative to verify maximum
\end{itemize}

\textbf{MLE Properties:}
\begin{itemize}
\item Invariance: If $\hat{\theta}$ is MLE of $\theta$, then $g(\hat{\theta})$ is MLE of $g(\theta)$
\item Consistent under regularity conditions
\item Asymptotically efficient
\item Asymptotic normality: $\hat{\theta}_n \xrightarrow{d} N(\theta, \frac{1}{nI(\theta)})$
\end{itemize}

\textbf{Fisher Information $I(\theta)$}

\textbf{Definition:} Measures information about $\theta$ from one observation.

Two equivalent forms:
\begin{enumerate}
\item Expected curvature: $I(\theta) = -\E\left[\frac{d^2\log f(X;\theta)}{d\theta^2}\right]$
\item Variance of score: $I(\theta) = \E\left[\left(\frac{d\log f(X;\theta)}{d\theta}\right)^2\right]$
\end{enumerate}

Relationship: Both equal under regularity conditions.

\textbf{How to compute $I(\theta)$:}
\begin{enumerate}
\item Write $\log f(x; \theta)$
\item Find score: $S(\theta) = \frac{d\log f}{d\theta}$
\item Method 1: $I(\theta) = \E[S(\theta)^2]$
\item Method 2: Find $\frac{d^2\log f}{d\theta^2}$, then $I(\theta) = -\E[\frac{d^2\log f}{d\theta^2}]$
\end{enumerate}

\textbf{Sample Fisher Information:}

For $n$ i.i.d. observations: $nI(\theta)$

Total log-likelihood: $\ell(\theta) = \sum_{i=1}^n \log f(X_i; \theta)$

Sample info: $I_n(\theta) = -\E\left[\frac{d^2\ell(\theta)}{d\theta^2}\right] = nI(\theta)$

\textbf{MLE Variance (Cramer-Rao):}

$$\Var(\hat{\theta}_{\text{MLE}}) \approx \frac{1}{nI(\theta)}$$

Estimated variance: $\widehat{\Var}(\hat{\theta}) = \frac{1}{nI(\hat{\theta})}$

\textbf{Common Fisher Information:}
\begin{itemize}
\item Bernoulli$(p)$: $I(p) = \frac{1}{p(1-p)}$
\item Poisson$(\lambda)$: $I(\lambda) = \frac{1}{\lambda}$
\item $N(\mu, \sigma^2)$ (known $\sigma^2$): $I(\mu) = \frac{1}{\sigma^2}$
\item $N(\mu, \sigma^2)$ (known $\mu$): $I(\sigma^2) = \frac{1}{2\sigma^4}$
\item Exponential$(\lambda)$: $I(\lambda) = \frac{1}{\lambda^2}$
\end{itemize}

\textbf{Example:} Bernoulli$(p)$ - Full derivation

\textit{Step 1: Single observation Fisher info}
\begin{align*}
f(x;p) &= p^x(1-p)^{1-x} \quad \text{(one obs)} \\
\log f &= x\log p + (1-x)\log(1-p) \\
\frac{d\log f}{dp} &= \frac{x}{p} - \frac{1-x}{1-p} \\
\frac{d^2\log f}{dp^2} &= -\frac{x}{p^2} - \frac{1-x}{(1-p)^2} \\
I(p) &= -\E\left[-\frac{X}{p^2} - \frac{1-X}{(1-p)^2}\right] \\
&= \frac{p}{p^2} + \frac{1-p}{(1-p)^2} = \frac{1}{p(1-p)}
\end{align*}

\textit{Step 2: Sample of size $n$}

For $X_1, \ldots, X_n$ i.i.d.: $nI(p) = \frac{n}{p(1-p)}$

\textit{Step 3: MLE variance}
$$\Var(\hat{p}) = \frac{1}{nI(p)} = \frac{p(1-p)}{n}$$

\textbf{Large Sample CI for MLE:}

Generic: $\hat{\theta} \pm z_{\alpha/2} \sqrt{\frac{1}{nI(\hat{\theta})}}$

For proportion: $\hat{p} \pm z_{\alpha/2}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}$

\vspace{2mm}
\fbox{\large\textbf{4. Confidence Intervals}}
\vspace{0.5mm}

\textbf{Definition}

$(1-\alpha)$100\% CI for $\theta$: Random interval $[L, U]$ with
$$P(L \le \theta \le U) = 1 - \alpha$$

Interpretation: $(1-\alpha)$100\% of intervals contain true $\theta$.

\textbf{Pivotal Quantity}

A function $Q(X_1, \ldots, X_n; \theta)$ whose distribution does not depend on $\theta$.

Examples: $\frac{\bar{X}-\mu}{\sigma/\sqrt{n}} \sim N(0,1)$, $\frac{\bar{X}-\mu}{S/\sqrt{n}} \sim t(n-1)$

To construct CI: Find $a, b$ with $P(a \le Q \le b) = 1-\alpha$, then solve for $\theta$.

\textbf{Common CIs}

\textbf{Mean ($\sigma$ known, large $n$):}
$$\bar{X} \pm z_{\alpha/2} \frac{\sigma}{\sqrt{n}}$$

\textbf{Mean ($\sigma$ unknown, Normal):}
$$\bar{X} \pm t_{\alpha/2, n-1} \frac{s}{\sqrt{n}}$$

\textbf{Proportion (large $n$):}
$$\hat{p} \pm z_{\alpha/2} \sqrt{\frac{\hat{p}(1-\hat{p})}{n}}$$

\textbf{Variance (Normal):}
$$\frac{(n-1)s^2}{\chi^2_{\alpha/2, n-1}} \le \sigma^2 \le \frac{(n-1)s^2}{\chi^2_{1-\alpha/2, n-1}}$$

\textbf{Ratio of Variances:}
$$\frac{s_1^2/s_2^2}{F_{\alpha/2, n_1-1, n_2-1}} \le \frac{\sigma_1^2}{\sigma_2^2} \le \frac{s_1^2}{s_2^2} \cdot F_{\alpha/2, n_1-1, n_2-1}$$

\textbf{Two-Sample Mean (equal var):}
$$(\bar{X}_1 - \bar{X}_2) \pm t_{\alpha/2, n_1+n_2-2} \cdot s_p\sqrt{\frac{1}{n_1} + \frac{1}{n_2}}$$

where $s_p^2 = \frac{(n_1-1)s_1^2 + (n_2-1)s_2^2}{n_1+n_2-2}$

\textbf{Sample Size Determination}

For mean with margin $E$ and confidence $(1-\alpha)$:
$$n = \left(\frac{z_{\alpha/2} \sigma}{E}\right)^2$$

For proportion:
$$n = \hat{p}(1-\hat{p})\left(\frac{z_{\alpha/2}}{E}\right)^2$$

Conservative (no prior $\hat{p}$): Use $\hat{p} = 0.5$ (gives largest $n$)

\fbox{\large\textbf{5. Hypothesis Testing}}
\vspace{0.5mm}

\textbf{Framework}

Null hypothesis $H_0$: Status quo (assume true)

Alternative $H_1$: What we want evidence for

Types:
\begin{itemize}
\item Two-sided: $H_0: \theta = \theta_0$ vs $H_1: \theta \neq \theta_0$
\item Upper: $H_0: \theta \le \theta_0$ vs $H_1: \theta > \theta_0$
\item Lower: $H_0: \theta \ge \theta_0$ vs $H_1: \theta < \theta_0$
\end{itemize}

\textbf{Error Types}

Type I Error (False Positive): $\alpha = P(\text{Reject } H_0 | H_0 \text{ true})$

Type II Error (False Negative): $\beta = P(\text{Fail reject } H_0 | H_1 \text{ true})$

Power: $1 - \beta = P(\text{Reject } H_0 | H_1 \text{ true})$

\textbf{p-Value}

Probability under $H_0$ of observing test statistic as extreme or more extreme.

Decision: Reject $H_0$ if $p \le \alpha$

Interpretation:
\begin{itemize}
\item $p \le 0.01$: Very strong evidence against $H_0$
\item $0.01 < p \le 0.05$: Moderate evidence
\item $0.05 < p \le 0.10$: Weak evidence
\item $p > 0.10$: Little/no evidence
\end{itemize}

\textbf{Common Test Statistics}

\textbf{Z-test (mean, $\sigma$ known):}
$$Z = \frac{\bar{X} - \mu_0}{\sigma/\sqrt{n}} \sim N(0,1) \text{ under } H_0$$

\textbf{t-test (mean, $\sigma$ unknown):}
$$t = \frac{\bar{X} - \mu_0}{s/\sqrt{n}} \sim t_{n-1} \text{ under } H_0$$

\textbf{Proportion test (large $n$):}
$$Z = \frac{\hat{p} - p_0}{\sqrt{p_0(1-p_0)/n}} \sim N(0,1) \text{ under } H_0$$

\textbf{Chi-square (variance):}
$$\chi^2 = \frac{(n-1)s^2}{\sigma_0^2} \sim \chi^2_{n-1} \text{ under } H_0$$

\textbf{Two-sample t-test (equal var):}
$$t = \frac{\bar{X}_1 - \bar{X}_2}{s_p\sqrt{1/n_1 + 1/n_2}} \sim t_{n_1+n_2-2} \text{ under } H_0$$
where $s_p^2 = \frac{(n_1-1)s_1^2 + (n_2-1)s_2^2}{n_1+n_2-2}$

\textbf{F-test (variance ratio):}
$$F = \frac{s_1^2}{s_2^2} \sim F_{n_1-1, n_2-1} \text{ under } H_0: \sigma_1^2 = \sigma_2^2$$

\textbf{Rejection Regions}

\textbf{Two-sided:}
\begin{itemize}
\item Z or t: $|T| > t_{\alpha/2}$
\item $\chi^2$: $\chi^2 < \chi^2_{1-\alpha/2}$ or $\chi^2 > \chi^2_{\alpha/2}$
\item F: $F < F_{1-\alpha/2}$ or $F > F_{\alpha/2}$
\end{itemize}

\textbf{One-sided (upper):}
\begin{itemize}
\item Z or t: $T > t_\alpha$
\item $\chi^2$: $\chi^2 > \chi^2_\alpha$
\item F: $F > F_\alpha$
\end{itemize}

\textbf{Power \& Sample Size}

\textit{Power function:} $\pi(\theta) = P(\text{Reject } H_0 | \theta)$

\textit{One-sided (upper):} $H_0: \mu \le \mu_0$ vs $H_1: \mu > \mu_0$
$$\pi(\mu_1) = \Phi\left(\frac{\mu_1 - \mu_0}{\sigma/\sqrt{n}} - z_\alpha\right)$$

\textit{Two-sided:} Replace $z_\alpha$ with $z_{\alpha/2}$

\textit{Sample size:} For power $1-\beta$ at $\mu_1$:
$$n = \left(\frac{(z_\alpha + z_\beta)\sigma}{\mu_1 - \mu_0}\right)^2$$

Two-sided: $n = \left(\frac{(z_{\alpha/2} + z_\beta)\sigma}{\mu_1 - \mu_0}\right)^2$

\textbf{Test Decision Rules}

\textit{p-value approach:}
\begin{itemize}
\item Calculate test statistic
\item Find p-value $= P(\text{as extreme or more} | H_0)$
\item Reject if $p \le \alpha$
\end{itemize}

\textit{Critical value approach:}
\begin{itemize}
\item Find critical value(s) from distribution
\item Reject if test statistic in rejection region
\end{itemize}

\textbf{p-Value Calculation}

\textit{Upper-tail:} $p = P(T \ge t_{\text{obs}} | H_0)$

\textit{Lower-tail:} $p = P(T \le t_{\text{obs}} | H_0)$

\textit{Two-tailed:} $p = 2P(|T| \ge |t_{\text{obs}}| | H_0)$

\textbf{CI-Testing Duality}

$(1-\alpha)$ CI $[L, U]$ corresponds to level-$\alpha$ test:
\begin{itemize}
\item Reject $H_0: \theta = \theta_0$ iff $\theta_0 \notin [L, U]$
\item Fail to reject iff $\theta_0 \in [L, U]$
\end{itemize}

\vspace{2mm}
\fbox{\large\textbf{6. Likelihood Ratio Tests}}
\vspace{0.5mm}

\textbf{Neyman-Pearson Lemma}

For simple $H_0: \theta = \theta_0$ vs simple $H_1: \theta = \theta_1$:

Most powerful test of size $\alpha$ rejects when:
$$\Lambda(x) = \frac{L(\theta_0; x)}{L(\theta_1; x)} \le k$$

where $k$ satisfies $P(\Lambda \le k | H_0) = \alpha$

\textbf{General LRT}

For composite hypotheses $H_0: \theta \in \Theta_0$ vs $H_1: \theta \in \Theta_1$:

\textbf{Parameter Spaces:}
\begin{itemize}
\item $\Omega$ = full parameter space (unrestricted)
\item $\Omega_0$ = restricted parameter space under $H_0$
\end{itemize}

Likelihood ratio:
$$\lambda(x) = \frac{\sup_{\theta \in \Omega_0} L(\theta; x)}{\sup_{\theta \in \Omega} L(\theta; x)} = \frac{L(\hat{\theta}_0; x)}{L(\hat{\theta}; x)}$$

where $\hat{\theta}_0$ = MLE under $H_0$ (restricted), $\hat{\theta}$ = MLE (unrestricted)

Properties: $0 \le \lambda(x) \le 1$, smaller $\lambda$ → stronger evidence against $H_0$

\textbf{Wilks' Theorem}

Under $H_0$ and regularity conditions:
$$-2\log\lambda(X) \xrightarrow{d} \chi^2_\nu$$

where $\nu = \dim(\Theta) - \dim(\Theta_0)$ (df = difference in parameters)

Rejection: $-2\log\lambda(x) > \chi^2_{\alpha, \nu}$

\textbf{LRT Calculation Steps}

\begin{enumerate}
\item Find unrestricted MLE: $\hat{\theta} = \arg\max_{\theta \in \Omega} L(\theta)$
\item Find restricted MLE: $\hat{\theta}_0 = \arg\max_{\theta \in \Omega_0} L(\theta)$
\item Compute: $\lambda = \frac{L(\hat{\theta}_0)}{L(\hat{\theta})}$
\item Test statistic: $-2\log\lambda$
\item Compare to $\chi^2_\nu$ critical value
\end{enumerate}

\textbf{Common LRT Scenarios}

\textit{Normal mean ($\sigma^2$ known):} $H_0: \mu = \mu_0$
\begin{itemize}
\item $\lambda = \exp\left\{-\frac{n(\bar{x}-\mu_0)^2}{2\sigma^2}\right\}$
\item $-2\log\lambda = \frac{n(\bar{x}-\mu_0)^2}{\sigma^2} \sim \chi^2_1$
\item Equivalent to Z-test
\end{itemize}

\textit{Normal variance ($\mu$ known):} $H_0: \sigma^2 = \sigma_0^2$
\begin{itemize}
\item $\lambda = \left(\frac{\hat{\sigma}^2}{\sigma_0^2}\right)^{n/2} e^{-\frac{n}{2}(\frac{\hat{\sigma}^2}{\sigma_0^2}-1)}$
\item $-2\log\lambda \sim \chi^2_1$
\end{itemize}

\textit{Two Normal means (equal var):} $H_0: \mu_1 = \mu_2$
\begin{itemize}
\item Simplifies to t-test
\item $\nu = 1$ (one constraint)
\end{itemize}

\textbf{LRT vs Other Tests}

\begin{itemize}
\item LRT is asymptotically optimal
\item For Normal, LRT = t-test or F-test
\item Use $\nu =$ \# free params (unrestricted) $-$ \# free params (restricted)
\end{itemize}

\textbf{P1 (2022):} $Y_1, \ldots, Y_n$ iid from $f(y) = \frac{1}{\theta}y^{\frac{1}{\theta}-1}$, $0 < y < 1$, $\theta > 0$. Find MLE of $\theta$.

Likelihood: $L(\theta) = \prod_{i=1}^n \frac{1}{\theta}y_i^{\frac{1}{\theta}-1} = \theta^{-n} \prod y_i^{\frac{1}{\theta}-1}$

Log-likelihood: $\ell(\theta) = -n\log\theta + (\frac{1}{\theta} - 1)\sum_{i=1}^n \log y_i$

Derivative: $\frac{d\ell}{d\theta} = -\frac{n}{\theta} - \frac{1}{\theta^2}\sum \log y_i = 0$

Solve: $n\theta = -\sum \log y_i$

$\boxed{\hat{\theta}_{\text{MLE}} = -\frac{1}{n}\sum_{i=1}^n \log Y_i = -\overline{\log Y}}$

\textbf{P2 (2022):} $Y_1, \ldots, Y_{25}$ iid $N(\mu, 9)$. Test $H_0: \mu = 18$ vs $H_1: \mu > 18$. Rejection region: $\bar{Y} > 19$. Find power when $\mu=20$.

Power $= P(\text{Reject } H_0 | \mu = 20)$

Under $\mu=20$: $\bar{Y} \sim N(20, 9/25)$, $\text{SE} = \sqrt{9/25} = 0.6$

Standardize: $P(\bar{Y} > 19) = P\left(\frac{\bar{Y}-20}{0.6} > \frac{19-20}{0.6}\right)$

$= P(Z > -1.667) = 1 - \Phi(-1.667) = \Phi(1.667)$

Using Z-table: $\boxed{\text{Power} = 0.9525}$

\textit{Note:} Find Type I error by evaluating at $\mu=18$

\textbf{P3 (2022):} $X_1, \ldots, X_m$ iid $N(\mu_X, 1)$, $Y_1, \ldots, Y_n$ iid $N(\mu_Y, 1)$ independent. Test $H_0: \mu_X - \mu_Y = 2$ vs $H_1: \mu_X - \mu_Y \neq 2$. Find rejection region $\bar{X} - \bar{Y} \notin [K_1, K_2]$ with Type I error $\alpha$.

Construct test statistic under $H_0$:

$\bar{X} - \bar{Y} \sim N(\mu_X - \mu_Y, \frac{1}{m} + \frac{1}{n})$

Under $H_0$: $\mu_X - \mu_Y = 2$

Standardize: $Z = \frac{(\bar{X}-\bar{Y}) - 2}{\sqrt{1/m + 1/n}} \sim N(0,1)$

Two-sided test: Reject if $|Z| \ge z_{\alpha/2}$, i.e., $\bar{X}-\bar{Y} \notin [K_1, K_2]$

$\boxed{K_1 = 2 - z_{\alpha/2}\sqrt{\frac{1}{m}+\frac{1}{n}}}$

$\boxed{K_2 = 2 + z_{\alpha/2}\sqrt{\frac{1}{m}+\frac{1}{n}}}$



\textbf{P4 (2022):} $Y_1, \ldots, Y_n$ iid $N(\mu, \sigma^2)$. Compare two estimators of $\mu$: $\bar{Y}$ and $\hat{\mu}^* = \frac{1}{n+1}[\sum Y_i + 10]$.

\textit{(a) Find $\Var(\bar{Y})$}

$\Var(\bar{Y}) = \Var\left(\frac{1}{n}\sum Y_i\right) = \frac{1}{n^2}\sum \Var(Y_i) = \frac{n\sigma^2}{n^2} = \boxed{\frac{\sigma^2}{n}}$

\textit{(b) Find $\Var(\hat{\mu}^*)$}

Rewrite: $\hat{\mu}^* = \frac{\sum Y_i + 10}{n+1} = \frac{n\bar{Y} + 10}{n+1}$

$\Var(\hat{\mu}^*) = \Var\left(\frac{n\bar{Y}}{n+1}\right) = \left(\frac{n}{n+1}\right)^2 \Var(\bar{Y})$

$= \frac{n^2}{(n+1)^2} \cdot \frac{\sigma^2}{n} = \boxed{\frac{n\sigma^2}{(n+1)^2}}$

Note: $\frac{n\sigma^2}{(n+1)^2} < \frac{\sigma^2}{n}$ since $\frac{n}{(n+1)^2} < \frac{1}{n}$

\textit{(c) Find MSE for both estimators}

For $\bar{Y}$: Unbiased, so MSE$(\bar{Y}) = \Var(\bar{Y}) = \frac{\sigma^2}{n}$

For $\hat{\mu}^*$: $\E[\hat{\mu}^*] = \frac{n\mu + 10}{n+1}$

Bias $= \frac{n\mu + 10}{n+1} - \mu = \frac{n\mu + 10 - (n+1)\mu}{n+1} = \frac{10-\mu}{n+1}$

MSE$(\hat{\mu}^*) = \Var(\hat{\mu}^*) + [\text{Bias}]^2 = \frac{n\sigma^2}{(n+1)^2} + \frac{(10-\mu)^2}{(n+1)^2}$

$= \boxed{\frac{n\sigma^2 + (10-\mu)^2}{(n+1)^2}}$

\textit{(d) Which is better if $\mu=10$?}

If $\mu=10$: Bias $= 0$, so MSE$(\hat{\mu}^*) = \frac{n\sigma^2}{(n+1)^2}$

Compare: $\frac{n\sigma^2}{(n+1)^2}$ vs $\frac{\sigma^2}{n}$

$\frac{n\sigma^2}{(n+1)^2} < \frac{\sigma^2}{n} \iff \frac{n^2}{(n+1)^2} < 1$ (always true)

$\implies \boxed{\hat{\mu}^* \text{ is better when } \mu=10}$

\textbf{P5 (2022):} $Y_1, \ldots, Y_n$ iid $N(\mu, \sigma^2)$. Consider MLE $\hat{\sigma}^2 = \frac{1}{n}\sum(Y_i-\bar{Y})^2$.

\textit{(a) Find $\E[\hat{\sigma}^2]$ and $\Var(\hat{\sigma}^2)$}

Expectation: Recall $(n-1)S^2 = \sum(Y_i-\bar{Y})^2$ where $S^2 = \frac{1}{n-1}\sum(Y_i-\bar{Y})^2$

Since $\E[S^2] = \sigma^2$:

$\E[\hat{\sigma}^2] = \E\left[\frac{1}{n}\sum(Y_i-\bar{Y})^2\right] = \frac{1}{n}\E[(n-1)S^2]$

$= \frac{n-1}{n}\E[S^2] = \boxed{\frac{n-1}{n}\sigma^2}$ (Biased! Underestimates by factor $\frac{n-1}{n}$)

Variance: Use fact that $\frac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)$

$\Var\left(\frac{(n-1)S^2}{\sigma^2}\right) = 2(n-1) \implies \Var(S^2) = \frac{2(n-1)\sigma^4}{(n-1)^2} = \frac{2\sigma^4}{n-1}$

$\Var(\hat{\sigma}^2) = \Var\left(\frac{n-1}{n}S^2\right) = \left(\frac{n-1}{n}\right)^2 \Var(S^2) = \boxed{\frac{2(n-1)\sigma^4}{n^2}}$

\textit{(b) Is $\hat{\sigma}^2$ consistent?}

Check: Bias $= \frac{n-1}{n}\sigma^2 - \sigma^2 = -\frac{\sigma^2}{n} \to 0$ as $n \to \infty$ ✓

$\Var(\hat{\sigma}^2) = \frac{2(n-1)\sigma^4}{n^2} \sim \frac{2\sigma^4}{n} \to 0$ as $n \to \infty$ ✓

Both bias and variance $\to 0$, so $\boxed{\text{YES, } \hat{\sigma}^2 \text{ is consistent}}$

\textbf{P6 (2022):} $Y \sim \text{Binomial}(n, p)$. MLE is $\hat{p} = Y/n$. Derive 95\% CI for $p$ using Fisher Information.

\textit{Step 1: Find Fisher Information}

For Bernoulli$(p)$: $I(p) = \frac{1}{p(1-p)}$ (single observation)

For $n$ observations: $nI(p) = \frac{n}{p(1-p)}$

\textit{Step 2: MLE variance (Cramér-Rao)}

$\Var(\hat{p}) \approx \frac{1}{nI(p)} = \frac{p(1-p)}{n}$

Standard error: $\text{SE}(\hat{p}) = \sqrt{\frac{p(1-p)}{n}}$

\textit{Step 3: Large sample CI}

Asymptotic normality: $\hat{p} \xrightarrow{d} N\left(p, \frac{p(1-p)}{n}\right)$

Standardize: $\frac{\hat{p} - p}{\sqrt{p(1-p)/n}} \xrightarrow{d} N(0,1)$

Plug-in estimate: Replace $p$ with $\hat{p}$ in SE

95\% CI: $\boxed{\hat{p} \pm z_{.025}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}} = \hat{p} \pm 1.96\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}}$

\textbf{P4 (2024):} $Y$ has single observation from $U(3, 3\theta)$. Test $H_0: \theta=4$ vs $H_1: \theta<4$. Rejection region: $Y < c$ with $\alpha=0.05$. Find $c$ and power at $\theta=3$.

\textit{Part (a): Find critical value $c$}

Under $H_0: \theta=4$: $Y \sim U(3, 12)$

CDF under $H_0$: $F(y) = \frac{y-3}{12-3} = \frac{y-3}{9}$ for $3 \le y \le 12$

Type I error: $\alpha = P(Y < c | H_0) = \frac{c-3}{9} = 0.05$

Solve: $c - 3 = 0.45 \implies \boxed{c = 3.45}$

\textit{Part (b): Calculate power at $\theta=3$}

Under $H_1: \theta=3$: $Y \sim U(3, 9)$

CDF under $H_1$: $F(y) = \frac{y-3}{9-3} = \frac{y-3}{6}$ for $3 \le y \le 9$

\textbf{P5 (2024):} Two independent populations: $X \sim N(\mu_1, 9)$ and $Y \sim N(\mu_2, 9)$. Samples: $n$ from pop 1, $2n$ from pop 2. Find minimum $n$ for 90\% CI of $\mu_1-\mu_2$ to have width $\le 3$.

\textit{Step 1: Find variance of estimator}

$\bar{X} - \bar{Y}$ is the estimator for $\mu_1 - \mu_2$

$\Var(\bar{X}-\bar{Y}) = \Var(\bar{X}) + \Var(\bar{Y}) = \frac{\sigma_1^2}{n} + \frac{\sigma_2^2}{2n}$

$= \frac{9}{n} + \frac{9}{2n} = \frac{18}{2n} + \frac{9}{2n} = \frac{27}{2n}$

\textit{Step 2: CI width formula}

90\% CI: $(\bar{X}-\bar{Y}) \pm z_{.05}\sqrt{\frac{27}{2n}}$ where $z_{.05} = 1.645$

Width $= 2z_{.05}\sqrt{\frac{27}{2n}} = 2(1.645)\sqrt{\frac{27}{2n}} = 3.29\sqrt{\frac{27}{2n}}$

\textit{Step 3: Solve inequality}

Require: $3.29\sqrt{\frac{27}{2n}} \le 3$

$\sqrt{\frac{27}{2n}} \le \frac{3}{3.29} = 0.912$

$\frac{27}{2n} \le (0.912)^2 = 0.8317$

$2n \ge \frac{27}{0.8317} = 32.46$

$n \ge 16.23$

Therefore: $\boxed{n = 17}$ (round up to nearest integer)
